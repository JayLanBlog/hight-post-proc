#version 460
// Gaussian Blur — fade-animated with uTime
layout(location=0) in vec2 vUV; layout(location=0) out vec4 outColor;
layout(binding=0) uniform sampler2D uInputTex;
layout(std140, binding=1) uniform Params {
    float uParamFloat0,uParamFloat1,uParamFloat2,uParamFloat3,uParamFloat4,uParamFloat5;
    vec2 uResolution; float uTime; float uFrameCount;
};
void main() {
    vec4 orig = texture(uInputTex, vUV);
    vec2 ts=uParamFloat0/uResolution;
vec4 blurred=texture(uInputTex,vUV)*0.4+texture(uInputTex,vUV+ts)*0.15+texture(uInputTex,vUV-ts)*0.15+texture(uInputTex,vUV+ts*2.0)*0.1+texture(uInputTex,vUV-ts*2.0)*0.1+texture(uInputTex,vUV+ts*6.0)*0.05+texture(uInputTex,vUV-ts*6.0)*0.05;
    float t = 0.5 + 0.5*sin(uTime*0.5);
    outColor = mix(orig, blurred, t);
}
